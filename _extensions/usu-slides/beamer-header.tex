% Navy title page, matching the HTML deck. \titlepage raises a one-shot flag; the
% background canvas paints navy while the flag is up and clears it at shipout, so
% every later frame keeps the normal background (standout frames keep theirs too).
\newif\ifnavytitle
\setbeamertemplate{background canvas}{%
  \ifnavytitle
    {\color{midnight}\vrule width\paperwidth height\paperheight}\global\navytitlefalse
  \else
    \ifbeamercolorempty[bg]{background canvas}{}{\color{bg}\vrule width\paperwidth height\paperheight}%
  \fi}
\let\metropolistitlepage\titlepage
\renewcommand{\titlepage}{%
  \global\navytitletrue
  \begingroup
    \setbeamercolor{normal text}{fg=offwhite}%
    \setbeamercolor{title}{fg=offwhite}%
    \setbeamercolor{subtitle}{fg=gold}%
    \setbeamercolor{author}{fg=offwhite}%
    \setbeamercolor{institute}{fg=offwhite}%
    \setbeamercolor{date}{fg=offwhite}%
    \usebeamercolor[fg]{normal text}%
    \metropolistitlepage
  \endgroup}
% Each `# Part ...` heading becomes a full navy standout frame, matching the HTML deck

% no bold Greek, so the symbols vanish. Route \mathbf through \symbf instead.
